 function renderContacts() {
  const box = $("contactsList");
  if (!box) return;

  const contacts = new Map();

  allMessages.forEach(m => {
    const phone = phoneOf(m);
    if (!phone) return;

    if (!contacts.has(phone)) {
      contacts.set(phone, {
        phone,
        lastMessage: "",
        lastTime: "",
        count: 0
      });
    }

    const c = contacts.get(phone);
    c.count++;

    const msg = m.text || m.message || m.body || "";
    if (msg) c.lastMessage = msg;
    c.lastTime = timeOf(m);
  });

  box.innerHTML = [...contacts.values()].map(c => `
    

      ${esc(c.phone)}
      
${esc(c.lastMessage || "No message text")}

      ${c.count} messages
    

  `).join("") || "
No contacts found.
";
}

function openChat(phone) {
  selectedPhone = digits(phone);

  const title = $("chatTitle");
  if (title) title.textContent = selectedPhone;

  const panel = $("chatPanel");
  if (panel) panel.style.display = "block";

  renderChatHistory();
}

function closeChat() {
  const panel = $("chatPanel");
  if (panel) panel.style.display = "none";
  selectedPhone = "";
}

function renderChatHistory() {
  const box = $("chatHistory");
  if (!box) return;

  const rows = allMessages
    .filter(m => digits(phoneOf(m)) === selectedPhone)
    .sort((a, b) => new Date(timeOf(a)) - new Date(timeOf(b)));

  box.innerHTML = rows.map(m => {
    const outgoing = directionOf(m) === "outgoing";
    const msg = m.text || m.message || m.body || "";
    return `
      

        
${esc(msg || "[Media or template message]")}

        ${esc(timeOf(m))}
      

    `;
  }).join("") || "
No message history loaded for this contact.
";
}

async function sendReply() {
  if (!API || !selectedPhone) {
    alert("Project API किंवा contact निवडलेला नाही.");
    return;
  }

  const input = $("replyText");
  const message = input?.value.trim();

  if (!message) {
    alert("Message लिहा.");
    return;
  }

  try {
    const result = await apiFetch("/send", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        to: selectedPhone,
        message
      })
    });

    if (result.error) throw new Error(result.error);

    input.value = "";
    status("chatStatus", "Reply request sent. Refreshing history...");
    await loadMessages();
  } catch (error) {
    status("chatStatus", "Reply failed: " + error.message, true);
  }
}

function showTemplateFields() {
  const select = $("templateName");
  const box = $("templateFields");
  if (!select || !box) return;

  const template = approvedTemplates.find(t =>
    (t.name || t.id) === select.value
  );

  if (!template) {
    box.innerHTML = "";
    return;
  }

  const bodyText = (template.components || [])
    .filter(c => c.type === "BODY")
    .map(c => c.text || "")
    .join(" ");

  const matches = bodyText.match(/\{\{\d+\}\}/g) || [];
  const count = new Set(matches).size;

  box.innerHTML = Array.from({ length: count }, (_, i) => `
    
    
  `).join("");
}

async function sendCampaign() {
  if (!API) {
    alert("आधी Project API सेट करा.");
    return;
  }

  const to = digits($("phoneNumber")?.value || "");
  const templateName = $("templateName")?.value || "";
  const language = $("templateLanguage")?.value || "en_US";
  const imageUrl = $("imageUrl")?.value.trim() || "";

  if (!to) {
    alert("ग्राहकाचा WhatsApp नंबर टाका.");
    return;
  }

  if (!templateName) {
    alert("Approved template निवडा.");
    return;
  }

  const template = approvedTemplates.find(t =>
    (t.name || t.id) === templateName
  );

  const bodyText = (template?.components || [])
    .filter(c => c.type === "BODY")
    .map(c => c.text || "")
    .join(" ");

  const count = new Set(bodyText.match(/\{\{\d+\}\}/g) || []).size;
  const parameters = [];

  for (let i = 1; i <= count; i++) {
    const value = $(`templateParam${i}`)?.value.trim() || "";
    if (!value) {
      alert(`Template parameter {{${i}}} भरा.`);
      return;
    }
    parameters.push(value);
  }

  try {
    status("sendResult", "Sending...");
    const result = await apiFetch("/send-template", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        to,
        templateName,
        language,
        imageUrl,
        parameters
      })
    });

    if (result.error) throw new Error(result.error);

    status("sendResult", "Request submitted. Provider response: " +
      JSON.stringify(result));
    await loadMessages();
  } catch (error) {
    status("sendResult", "Send failed: " + error.message, true);
  }
}

function saveLead() {
  const phone = digits($("leadPhone")?.value || "");
  const name = $("leadName")?.value.trim() || "";
  const note = $("leadNote")?.value.trim() || "";

  if (!phone) {
    alert("Lead चा फोन नंबर टाका.");
    return;
  }

  const leads = JSON.parse(
    localStorage.getItem("ak_property_leads") || "[]"
  );

  leads.push({
    phone,
    name,
    note,
    project: activeProject,
    createdAt: new Date().toISOString()
  });

  localStorage.setItem("ak_property_leads", JSON.stringify(leads));
  alert("Lead या browser मध्ये save झाला.");
  showLeads();
  updateDashboard();
}

function showLeads() {
  const box = $("leadsList");
  if (!box) return;

  const leads = JSON.parse(
    localStorage.getItem("ak_property_leads") || "[]"
  ).filter(l => l.project === activeProject);

  box.innerHTML = leads.map(l => `
    

      ${esc(l.name || "Unnamed Lead")}
      
${esc(l.phone)}

      
${esc(l.note || "")}

      ${esc(l.createdAt)}
    

  `).join("") || "
No leads saved for this project.
";
}

async function loadReports() {
  if (!API) {
    status("reportsStatus", "Project API सेट केलेली नाही.", true);
    return;
  }

  try {
    const result = await apiFetch("/reports?limit=1000&offset=0");
    reportRows = result.messages || result.data || [];

    const summary = result.summary || {};
    const box = $("reportsContent");

    if (box) {
      box.innerHTML = `
        
Total messages: ${Number(summary.total ?? totalMessages)}

        
Outgoing: ${Number(summary.outgoing ?? totalOutgoing)}

        
Incoming: ${Number(summary.incoming ?? totalIncoming)}

      `;
    }

    status("reportsStatus", "Reports loaded.");
  } catch (error) {
    status("reportsStatus", "Reports failed: " + error.message, true);
  }
}

function exportReports() {
  const rows = reportRows.length ? reportRows : allMessages;

  if (!rows.length) {
    alert("Export करण्यासाठी report data उपलब्ध नाही.");
    return;
  }

  const columns = ["id", "to", "from", "phone", "direction",
                   "status", "text", "message", "created_at", "timestamp"];

  const csvCell = value =>
    '"' + String(value ?? "").replace(/"/g, '""') + '"';

  const csv = [
    columns.join(","),
    ...rows.map(row =>
      columns.map(key => csvCell(row[key])).join(",")
    )
  ].join("\r\n");

  const blob = new Blob(["\uFEFF" + csv], {
    type: "text/csv;charset=utf-8;"
  });

  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = `${activeProject}-whatsapp-reports.csv`;
  a.click();
  URL.revokeObjectURL(url);
}
